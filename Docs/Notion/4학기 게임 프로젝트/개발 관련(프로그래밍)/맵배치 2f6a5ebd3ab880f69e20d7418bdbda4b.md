# 맵배치

현재 파이썬 코드는 .scene를 뽑아내는 코드임. 배치를 gpt한테 맡기자는 이야기임

밑의 파이썬 코드를 gpt 갈궈서 어떤거 어떤 위치로 보내고 어떻게 배치하라고 부탁하면 해줌. 

- 파이썬 코드
    
    ```python
    import json
    import math
    
    def deg(v: float) -> float:
        """엔진 .scene rotation이 라디안이라면, 각도(deg) -> 라디안(rad)"""
        return math.radians(v)
    
    def generate_scene():
        entities = []
        guid = 10000
    
        # ==========================================
        # [노출] 에셋 이름/경로 한 곳에서 관리
        ASSETS = {
            "Tile": {"path": "Assets/Fbx/tile_mix.fbxasset", "mesh": "tile_mix"},
            "Pillar": {"path": "Assets/Fbx/pillar.fbxasset", "mesh": "pillar_low"},
            "DoorFrame": {"path": "Assets/Fbx/doorframe.fbxasset", "mesh": "doorframe_low"},
            "Door": {"path": "Assets/Fbx/door.fbxasset", "mesh": "door_low"},
            "Boss": {"path": "Assets/Fbx/boss.fbxasset", "mesh": "boss"},
        }
        # ==========================================
    
        # ==========================================
        # [노출] 배치 파라미터
        # ✅ 타일 pivot = "왼쪽-위(top-left corner)" 가정
        TILE_SPACING = 2.0
    
        # ✅ 바닥 X: 왼쪽 7칸 + 오른쪽 7칸 = 14칸
        FLOOR_HALF_TILES_X = 7
        map_width = FLOOR_HALF_TILES_X * 2  # 14
    
        # Z 길이
        map_length = 28
    
        # ✅ 벽 높이: 4칸
        WALL_STACK_COUNT = 4
    
        # ✅ 천장 높이
        CEILING_Y = WALL_STACK_COUNT * TILE_SPACING  # 예: 8.0
    
        # 문 구멍(타일 1칸) 인덱스
        DOOR_HOLE_X_INDEX = 6
    
        # 벽 쌓기: 위에서부터
        WALL_STACK_FROM_TOP = True
    
        # ✅ 벽들을 아래로 1칸 내리기
        WALL_Y_OFFSET = -TILE_SPACING
    
        # ✅ 요청 반영 오프셋들
        FLOOR_Z_OFFSET = +2.0     # 바닥 z +2
        CEILING_Z_OFFSET = +2.0   # 천장 z +2
        BACK_WALL_X_OFFSET = +2.0 # 뒷벽 x +2
    
        # ✅ 최종 요청: 왼쪽 벽만 z 방향으로 +2
        LEFT_WALL_Z_OFFSET = +2.0
        # ==========================================
    
        # ------------------------------------------------------------
        # ✅ top-left pivot 배치를 위한 "맵 시작점"
        # ------------------------------------------------------------
        map_width_world = map_width * TILE_SPACING
        map_length_world = map_length * TILE_SPACING
    
        map_x_start = -map_width_world * 0.5
        map_z_start = -map_length_world * 0.5
        map_x_end = map_x_start + map_width_world
        map_z_end = map_z_start + map_length_world
    
        # ------------------------------------------------------------
        # 1. 메인 카메라
        # ------------------------------------------------------------
        guid += 1
        entities.append({
            "Camera": {"FOV": 60.0, "Far Plane": 1000.0, "Near Plane": 0.1, "primary": True},
            "Transform": {
                "enabled": True,
                "position": {"x": 0.0, "y": 15.0, "z": -35.0},
                "rotation": {"x": deg(35.0), "y": deg(0.0), "z": deg(0.0)},
                "scale": {"x": 1.0, "y": 1.0, "z": 1.0},
            },
            "guid": str(guid),
            "name": "MainCamera",
        })
    
        # ------------------------------------------------------------
        # 2. 바닥/천장  ✅ z를 +2 이동
        # ------------------------------------------------------------
        for z in range(map_length):
            for x in range(map_width):
                pos_x = map_x_start + (x * TILE_SPACING)
                pos_z = map_z_start + (z * TILE_SPACING)
    
                # 바닥 (z + 2)
                guid += 1
                entities.append({
                    "SkinnedMesh": {
                        "boneCount": 0,
                        "instanceAssetPath": ASSETS["Tile"]["path"],
                        "meshAssetPath": ASSETS["Tile"]["mesh"],
                    },
                    "Transform": {
                        "enabled": True,
                        "position": {"x": pos_x, "y": 0.0, "z": pos_z + FLOOR_Z_OFFSET},
                        "rotation": {"x": 0.0, "y": 0.0, "z": 0.0},
                        "scale": {"x": 1.0, "y": 1.0, "z": 1.0},
                    },
                    "guid": str(guid),
                    "name": f"Floor_{x}_{z}",
                })
    
                # 천장 (z + 2)
                guid += 1
                entities.append({
                    "SkinnedMesh": {
                        "boneCount": 0,
                        "instanceAssetPath": ASSETS["Tile"]["path"],
                        "meshAssetPath": ASSETS["Tile"]["mesh"],
                    },
                    "Transform": {
                        "enabled": True,
                        "position": {"x": pos_x, "y": CEILING_Y, "z": pos_z + CEILING_Z_OFFSET},
                        "rotation": {"x": 0.0, "y": 0.0, "z": 0.0},
                        "scale": {"x": 1.0, "y": 1.0, "z": 1.0},
                    },
                    "guid": str(guid),
                    "name": f"Ceiling_{x}_{z}",
                })
    
        # ------------------------------------------------------------
        # 3. 기둥 (바닥이 z+2 이동했으니 기둥도 같이 맞춰줌)
        # ------------------------------------------------------------
        pillar_indices = [0, map_width - 1]
        for z in range(0, map_length, 4):
            tile_z = map_z_start + (z * TILE_SPACING)
            for x_idx in pillar_indices:
                tile_x = map_x_start + (x_idx * TILE_SPACING)
    
                pos_x = tile_x + (TILE_SPACING * 0.5)
                pos_z = tile_z + (TILE_SPACING * 0.5) + FLOOR_Z_OFFSET
    
                guid += 1
                entities.append({
                    "SkinnedMesh": {
                        "boneCount": 0,
                        "instanceAssetPath": ASSETS["Pillar"]["path"],
                        "meshAssetPath": ASSETS["Pillar"]["mesh"],
                    },
                    "Transform": {
                        "enabled": True,
                        "position": {"x": pos_x, "y": 0.0, "z": pos_z},
                        "rotation": {"x": 0.0, "y": 0.0, "z": 0.0},
                        "scale": {"x": 1.0, "y": 1.0, "z": 1.0},
                    },
                    "guid": str(guid),
                    "name": f"Pillar_{x_idx}_{z}",
                })
    
        # ------------------------------------------------------------
        # 3.5. 벽 (아래로 1칸 내림: WALL_Y_OFFSET 적용)
        # ------------------------------------------------------------
        RX_WALL = deg(90.0)
    
        def wall_layer_y(layer_index: int) -> float:
            if WALL_STACK_FROM_TOP:
                return (CEILING_Y - (layer_index * TILE_SPACING)) + WALL_Y_OFFSET
            else:
                return (TILE_SPACING * (layer_index + 1)) + WALL_Y_OFFSET
    
        WALL_LEFT_X = map_x_start
        WALL_RIGHT_X = map_x_end
        WALL_FRONT_Z = map_z_start
        WALL_BACK_Z = map_z_end
    
        # 앞 벽 (문 구멍 1칸)
        for layer in range(WALL_STACK_COUNT):
            wy = wall_layer_y(layer)
            for x in range(map_width):
                if x == DOOR_HOLE_X_INDEX:
                    continue
                pos_x = map_x_start + (x * TILE_SPACING)
                guid += 1
                entities.append({
                    "SkinnedMesh": {"boneCount": 0, "instanceAssetPath": ASSETS["Tile"]["path"], "meshAssetPath": ASSETS["Tile"]["mesh"]},
                    "Transform": {
                        "enabled": True,
                        "position": {"x": pos_x, "y": wy, "z": WALL_FRONT_Z},
                        "rotation": {"x": RX_WALL, "y": deg(0.0), "z": deg(0.0)},
                        "scale": {"x": 1.0, "y": 1.0, "z": 1.0},
                    },
                    "guid": str(guid),
                    "name": f"Wall_Front_L{layer}_{x}",
                })
    
        # 뒤 벽 ✅ x를 +2 이동
        for layer in range(WALL_STACK_COUNT):
            wy = wall_layer_y(layer)
            for x in range(map_width):
                pos_x = map_x_start + (x * TILE_SPACING) + BACK_WALL_X_OFFSET
                guid += 1
                entities.append({
                    "SkinnedMesh": {"boneCount": 0, "instanceAssetPath": ASSETS["Tile"]["path"], "meshAssetPath": ASSETS["Tile"]["mesh"]},
                    "Transform": {
                        "enabled": True,
                        "position": {"x": pos_x, "y": wy, "z": WALL_BACK_Z},
                        "rotation": {"x": RX_WALL, "y": deg(180.0), "z": deg(0.0)},
                        "scale": {"x": 1.0, "y": 1.0, "z": 1.0},
                    },
                    "guid": str(guid),
                    "name": f"Wall_Back_L{layer}_{x}",
                })
    
        # 왼쪽 벽 ✅ z를 +2 이동
        for layer in range(WALL_STACK_COUNT):
            wy = wall_layer_y(layer)
            for z in range(map_length):
                pos_z = map_z_start + (z * TILE_SPACING)
                guid += 1
                entities.append({
                    "SkinnedMesh": {"boneCount": 0, "instanceAssetPath": ASSETS["Tile"]["path"], "meshAssetPath": ASSETS["Tile"]["mesh"]},
                    "Transform": {
                        "enabled": True,
                        "position": {"x": WALL_LEFT_X, "y": wy, "z": pos_z + LEFT_WALL_Z_OFFSET},
                        "rotation": {"x": RX_WALL, "y": deg(90.0), "z": deg(0.0)},
                        "scale": {"x": 1.0, "y": 1.0, "z": 1.0},
                    },
                    "guid": str(guid),
                    "name": f"Wall_Left_L{layer}_{z}",
                })
    
        # 오른쪽 벽
        for layer in range(WALL_STACK_COUNT):
            wy = wall_layer_y(layer)
            for z in range(map_length):
                pos_z = map_z_start + (z * TILE_SPACING)
                guid += 1
                entities.append({
                    "SkinnedMesh": {"boneCount": 0, "instanceAssetPath": ASSETS["Tile"]["path"], "meshAssetPath": ASSETS["Tile"]["mesh"]},
                    "Transform": {
                        "enabled": True,
                        "position": {"x": WALL_RIGHT_X, "y": wy, "z": pos_z},
                        "rotation": {"x": RX_WALL, "y": deg(-90.0), "z": deg(0.0)},
                        "scale": {"x": 1.0, "y": 1.0, "z": 1.0},
                    },
                    "guid": str(guid),
                    "name": f"Wall_Right_L{layer}_{z}",
                })
    
        # ------------------------------------------------------------
        # 4. 문
        # ------------------------------------------------------------
        door_x = (map_x_start + (DOOR_HOLE_X_INDEX * TILE_SPACING)) + (TILE_SPACING * 0.5)
        door_z = WALL_FRONT_Z
    
        guid += 1
        entities.append({
            "SkinnedMesh": {"boneCount": 0, "instanceAssetPath": ASSETS["DoorFrame"]["path"], "meshAssetPath": ASSETS["DoorFrame"]["mesh"]},
            "Transform": {
                "enabled": True,
                "position": {"x": door_x, "y": 0.0, "z": door_z},
                "rotation": {"x": 0.0, "y": 0.0, "z": 0.0},
                "scale": {"x": 1.0, "y": 1.0, "z": 1.0},
            },
            "guid": str(guid),
            "name": "DoorFrame",
        })
    
        guid += 1
        entities.append({
            "SkinnedMesh": {"boneCount": 0, "instanceAssetPath": ASSETS["Door"]["path"], "meshAssetPath": ASSETS["Door"]["mesh"]},
            "Transform": {
                "enabled": True,
                "position": {"x": door_x, "y": 0.0, "z": door_z},
                "rotation": {"x": 0.0, "y": 0.0, "z": 0.0},
                "scale": {"x": 1.0, "y": 1.0, "z": 1.0},
            },
            "guid": str(guid),
            "name": "Door",
        })
    
        # ------------------------------------------------------------
        # 5. 보스 (바닥이 z+2 이동했으니 보스도 같이 맞춰줌)
        # ------------------------------------------------------------
        boss_x = 0.0
        boss_z = (map_z_end - (TILE_SPACING * 2.0)) + FLOOR_Z_OFFSET
    
        guid += 1
        entities.append({
            "SkinnedMesh": {"boneCount": 0, "instanceAssetPath": ASSETS["Boss"]["path"], "meshAssetPath": ASSETS["Boss"]["mesh"]},
            "Transform": {
                "enabled": True,
                "position": {"x": boss_x, "y": 0.0, "z": boss_z},
                "rotation": {"x": deg(0.0), "y": deg(180.0), "z": deg(0.0)},
                "scale": {"x": 1.0, "y": 1.0, "z": 1.0},
            },
            "guid": str(guid),
            "name": "Boss",
        })
    
        # 저장
        scene_data = {"entities": entities, "sceneName": "PrototypeDungeon", "version": 1}
        output_filename = "PrototypeDungeon.scene"
        with open(output_filename, "w", encoding="utf-8") as f:
            json.dump(scene_data, f, indent=4)
    
        print(
            f"성공: '{output_filename}' 생성 완료 "
            f"(spacing={TILE_SPACING}, width={map_width}, length={map_length}, walls={WALL_STACK_COUNT}층, "
            f"floorZ+={FLOOR_Z_OFFSET}, ceilingZ+={CEILING_Z_OFFSET}, backWallX+={BACK_WALL_X_OFFSET}, "
            f"leftWallZ+={LEFT_WALL_Z_OFFSET}, entities={len(entities)})"
        )
    
    if __name__ == "__main__":
        generate_scene()
    
    ```
