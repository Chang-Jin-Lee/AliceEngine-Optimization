    ```csharp
            public static void BakeAll(VfxBakeSettings settings)
            {
                var folders = SplitList(settings.scanFolders).ToArray();
                if (folders.Length == 0)
                    folders = new[] { "Assets" };
    
                var exclude = SplitList(settings.excludeKeywords).ToArray();
                var guids = AssetDatabase.FindAssets("t:Prefab", folders);
    
                int total = guids.Length;
                int baked = 0;
                int failed = 0;
    
                var manifest = new List<ManifestEntry>();
    
                try
                {
                    for (int i = 0; i < total; ++i)
                    {
                        var path = AssetDatabase.GUIDToAssetPath(guids[i]);
                        if (IsExcluded(path, exclude))
                            continue;
    
                        EditorUtility.DisplayProgressBar("VFX Bake", path, (float)i / Math.Max(1, total));
    
                        try
                        {
                            if (BakePrefab(path, settings, out var entry))
                            {
                                baked++;
                                if (entry != null)
                                    manifest.Add(entry);
                            }
                        }
                        catch (Exception e)
                        {
                            failed++;
                            Debug.LogError($"[VFX Bake] Failed: {path} ({e.Message})");
                        }
                    }
                }
                finally
                {
                    EditorUtility.ClearProgressBar();
                }
    
                if (settings.writeManifest)
                    WriteManifest(settings.outputRoot, manifest);
    
                EditorUtility.DisplayDialog("VFX Bake", $"Scan: {total}\nBaked: {baked}\nFailed: {failed}", "OK");
            }
    
            public static void BakeSelected(VfxBakeSettings settings, UnityEngine.Object selected)
            {
                if (selected == null)
                {
                    EditorUtility.DisplayDialog("VFX Bake", "No prefab selected.", "OK");
                    return;
                }
    
                var path = AssetDatabase.GetAssetPath(selected);
                if (string.IsNullOrEmpty(path) || !path.EndsWith(".prefab", StringComparison.OrdinalIgnoreCase))
                {
                    EditorUtility.DisplayDialog("VFX Bake", "Select a prefab asset.", "OK");
                    return;
                }
    
                try
                {
                    if (BakePrefab(path, settings, out var entry))
                    {
                        if (settings.writeManifest && entry != null)
                            WriteManifest(settings.outputRoot, new List<ManifestEntry> { entry });
                        EditorUtility.DisplayDialog("VFX Bake", "Bake complete.", "OK");
                    }
                    else
                    {
                        EditorUtility.DisplayDialog("VFX Bake", "Prefab has no ParticleSystem/VFX.", "OK");
                    }
                }
                catch (Exception e)
                {
                    EditorUtility.DisplayDialog("VFX Bake", $"Bake failed: {e.Message}", "OK");
                }
            }
    
            private static bool BakePrefab(string prefabPath, VfxBakeSettings settings, out ManifestEntry entry)
            {
                entry = null;
                var prefab = AssetDatabase.LoadAssetAtPath<GameObject>(prefabPath);
                if (prefab == null)
                    return false;
    
                var prevScene = SceneManager.GetActiveScene();
                var tempScene = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Additive);
                EditorSceneManager.SetActiveScene(tempScene);
    
                GameObject root = null;
                try
                {
                    root = (GameObject)PrefabUtility.InstantiatePrefab(prefab, tempScene);
                    if (root == null)
                        return false;
    
                    root.transform.position = Vector3.zero;
                    root.transform.rotation = Quaternion.identity;
                    root.transform.localScale = Vector3.one;
    
                    var systems = root.GetComponentsInChildren<ParticleSystem>(true);
                    if (systems == null || systems.Length == 0)
                        return false;
    
                    var roots = GetRootSystems(systems);
    
                    var seedBackup = new List<(ParticleSystem ps, bool autoSeed, uint seed)>();
                    if (settings.fixedSeed)
                    {
                        uint s = settings.seed;
                        foreach (var ps in systems)
                        {
                            seedBackup.Add((ps, ps.useAutoRandomSeed, ps.randomSeed));
                            ps.useAutoRandomSeed = false;
                            ps.randomSeed = s++;
                        }
                    }
    
                    float duration = settings.useAutoDuration ? ComputeAutoDuration(systems) : settings.duration;
                    duration = Mathf.Max(0.01f, duration);
                    float fps = Mathf.Max(1.0f, settings.fps);
                    int frameCount = Mathf.Max(1, Mathf.CeilToInt(duration * fps));
                    int tilesX, tilesY;
                    ComputeGrid(frameCount, out tilesX, out tilesY);
    
                    var bounds = ComputeSimulatedBounds(root, roots, frameCount, 1.0f / fps);
    
                    var camGO = new GameObject("__VfxBakeCamera");
                    var cam = camGO.AddComponent<Camera>();
                    SetupCamera(cam, bounds, settings);
    
                    var rt = new RenderTexture(settings.tileWidth, settings.tileHeight, 24, RenderTextureFormat.ARGB32);
                    rt.antiAliasing = 1;
                    cam.targetTexture = rt;
    
                    var frameTex = new Texture2D(settings.tileWidth, settings.tileHeight, TextureFormat.RGBA32, false);
                    var sheet = new Texture2D(settings.tileWidth * tilesX, settings.tileHeight * tilesY, TextureFormat.RGBA32, false);
    
                    ResetSimulation(roots);
                    for (int i = 0; i < frameCount; ++i)
                    {
                        if (i > 0)
                            StepSimulation(roots, 1.0f / fps);
    
                        cam.Render();
                        RenderTexture.active = rt;
                        frameTex.ReadPixels(new Rect(0, 0, settings.tileWidth, settings.tileHeight), 0, 0);
                        frameTex.Apply();
    
                        int col = i % tilesX;
                        int row = i / tilesX;
                        int x = col * settings.tileWidth;
                        int y = (tilesY - 1 - row) * settings.tileHeight; // top-to-bottom
                        var pixels = frameTex.GetPixels();
                        sheet.SetPixels(x, y, settings.tileWidth, settings.tileHeight, pixels);
                    }
    
                    sheet.Apply();
                    RenderTexture.active = null;
    
                    string guid = AssetDatabase.AssetPathToGUID(prefabPath);
                    string effectName = SanitizeName(Path.GetFileNameWithoutExtension(prefabPath));
                    string effectFolder = Path.Combine(settings.outputRoot, effectName);
                    if (Directory.Exists(effectFolder))
                        effectFolder = Path.Combine(settings.outputRoot, $"{effectName}_{guid.Substring(0, 8)}");
    
                    Directory.CreateDirectory(effectFolder);
    
                    string pngPath = Path.Combine(effectFolder, "flipbook.png");
                    File.WriteAllBytes(pngPath, sheet.EncodeToPNG());
    
                    var jsonPath = Path.Combine(effectFolder, "flipbook.json");
                    var meta = new JObject
                    {
                        ["version"] = 1,
                        ["effectName"] = effectName,
                        ["sourcePrefabPath"] = prefabPath,
                        ["sourcePrefabGuid"] = guid,
                        ["unityVersion"] = Application.unityVersion,
                        ["texturePath"] = ToLogicalPath(pngPath),
                        ["tilesX"] = tilesX,
                        ["tilesY"] = tilesY,
                        ["frameCount"] = frameCount,
                        ["fps"] = fps,
                        ["duration"] = duration,
                        ["rowOrder"] = "TopToBottom",
                        ["tileSize"] = new JArray(settings.tileWidth, settings.tileHeight),
                        ["sheetSize"] = new JArray(settings.tileWidth * tilesX, settings.tileHeight * tilesY),
                        ["size"] = new JArray(bounds.size.x, bounds.size.y),
                        ["pivot"] = new JArray(0.0f, 0.0f)
                    };
    
                    File.WriteAllText(jsonPath, meta.ToString(Formatting.Indented));
    
                    entry = new ManifestEntry
                    {
                        name = effectName,
                        flipbookPath = ToLogicalPath(jsonPath),
                        texturePath = ToLogicalPath(pngPath),
                        tilesX = tilesX,
                        tilesY = tilesY,
                        frameCount = frameCount,
                        fps = fps,
                        size = new Vector2(bounds.size.x, bounds.size.y),
                        pivot = Vector2.zero
                    };
    
                    if (settings.fixedSeed)
                    {
                        foreach (var s in seedBackup)
                        {
                            s.ps.useAutoRandomSeed = s.autoSeed;
                            s.ps.randomSeed = s.seed;
                        }
                    }
    
                    UnityEngine.Object.DestroyImmediate(camGO);
                    UnityEngine.Object.DestroyImmediate(frameTex);
                    UnityEngine.Object.DestroyImmediate(sheet);
                    UnityEngine.Object.DestroyImmediate(rt);
                    return true;
                }
                finally
                {
                    if (root != null)
                        UnityEngine.Object.DestroyImmediate(root);
                    EditorSceneManager.SetActiveScene(prevScene);
                    EditorSceneManager.CloseScene(tempScene, true);
                }
            }
    
            private static void WriteManifest(string outputRoot, List<ManifestEntry> entries)
            {
                if (entries == null || entries.Count == 0)
                    return;
    
                var list = new JArray();
                foreach (var e in entries)
                {
                    var obj = new JObject
                    {
                        ["name"] = e.name,
                        ["flipbookPath"] = e.flipbookPath,
                        ["texturePath"] = e.texturePath,
                        ["tilesX"] = e.tilesX,
                        ["tilesY"] = e.tilesY,
                        ["frameCount"] = e.frameCount,
                        ["fps"] = e.fps,
                        ["rowOrder"] = "TopToBottom",
                        ["size"] = new JArray(e.size.x, e.size.y),
                        ["pivot"] = new JArray(e.pivot.x, e.pivot.y)
                    };
                    list.Add(obj);
                }
    
                var root = new JObject
                {
                    ["version"] = 1,
                    ["generatedAt"] = DateTime.Now.ToString("yyyy-MM-dd HH:mm:ss"),
                    ["effects"] = list
                };
    
                var manifestPath = Path.Combine(outputRoot, "manifest.json");
                File.WriteAllText(manifestPath, root.ToString(Formatting.Indented));
            }
    
            private static List<ParticleSystem> GetRootSystems(ParticleSystem[] systems)
            {
                var list = new List<ParticleSystem>();
                var set = new HashSet<ParticleSystem>(systems);
                foreach (var ps in systems)
                {
                    bool hasParent = false;
                    var t = ps.transform.parent;
                    while (t != null)
                    {
                        var parentPs = t.GetComponent<ParticleSystem>();
                        if (parentPs != null && set.Contains(parentPs))
                        {
                            hasParent = true;
                            break;
                        }
                        t = t.parent;
                    }
                    if (!hasParent)
                        list.Add(ps);
                }
                return list;
            }
    
            private static void ResetSimulation(List<ParticleSystem> roots)
            {
                foreach (var ps in roots)
                    ps.Simulate(0f, true, true, true);
            }
    
            private static void StepSimulation(List<ParticleSystem> roots, float dt)
            {
                foreach (var ps in roots)
                    ps.Simulate(dt, true, false, true);
            }
    
            private static float ComputeAutoDuration(ParticleSystem[] systems)
            {
                float maxDuration = 0.0f;
                foreach (var ps in systems)
                {
                    var main = ps.main;
                    float lifetime = main.startLifetime.constantMax;
                    maxDuration = Mathf.Max(maxDuration, main.duration + lifetime);
                }
                return Mathf.Max(0.1f, maxDuration);
            }
    
            private static Bounds ComputeSimulatedBounds(GameObject root, List<ParticleSystem> roots, int frameCount, float dt)
            {
                bool hasBounds = false;
                Bounds b = new Bounds(Vector3.zero, Vector3.one);
    
                ResetSimulation(roots);
                for (int i = 0; i < frameCount; ++i)
                {
                    if (i > 0)
                        StepSimulation(roots, dt);
    
                    foreach (var r in root.GetComponentsInChildren<Renderer>(true))
                    {
                        if (!hasBounds)
                        {
                            b = r.bounds;
                            hasBounds = true;
                        }
                        else
                        {
                            b.Encapsulate(r.bounds);
                        }
                    }
                }
    
                if (!hasBounds)
                    b = new Bounds(Vector3.zero, Vector3.one);
    
                return b;
            }
    
            private static void SetupCamera(Camera cam, Bounds bounds, VfxBakeSettings settings)
            {
                cam.clearFlags = CameraClearFlags.SolidColor;
                cam.backgroundColor = new Color(0, 0, 0, 0);
                cam.orthographic = settings.orthographic;
                cam.nearClipPlane = 0.01f;
                cam.farClipPlane = 1000.0f;
    
                Vector3 center = bounds.center;
                float radius = Mathf.Max(bounds.extents.x, bounds.extents.y, bounds.extents.z);
                float pad = Mathf.Max(0.0f, settings.boundsPadding);
                radius *= (1.0f + pad);
    
                if (settings.orthographic)
                {
                    cam.orthographicSize = Mathf.Max(0.1f, radius);
                    cam.transform.position = center + new Vector3(0, 0, -Mathf.Max(2.0f, radius * 2.0f));
                    cam.transform.rotation = Quaternion.identity;
                }
                else
                {
                    cam.fieldOfView = Mathf.Max(1.0f, settings.cameraFov);
                    float dist = radius / Mathf.Tan(cam.fieldOfView * 0.5f * Mathf.Deg2Rad);
                    cam.transform.position = center + new Vector3(0, 0, -Mathf.Max(2.0f, dist));
                    cam.transform.rotation = Quaternion.identity;
                }
    
                cam.transform.LookAt(center, Vector3.up);
            }
    
            private static void ComputeGrid(int frameCount, out int tilesX, out int tilesY)
            {
                tilesX = Mathf.CeilToInt(Mathf.Sqrt(frameCount));
                tilesY = Mathf.CeilToInt((float)frameCount / tilesX);
                tilesX = Mathf.Max(1, tilesX);
                tilesY = Mathf.Max(1, tilesY);
            }
    
            private static IEnumerable<string> SplitList(string s)
            {
                if (string.IsNullOrEmpty(s))
                    yield break;
    
                var parts = s.Split(new[] { ',', ';', '\n', '\r' }, StringSplitOptions.RemoveEmptyEntries);
                foreach (var p in parts)
                {
                    var trimmed = p.Trim();
                    if (!string.IsNullOrEmpty(trimmed))
                        yield return trimmed;
                }
            }
    
            private static bool IsExcluded(string path, string[] keywords)
            {
                if (keywords == null || keywords.Length == 0)
                    return false;
    
                string lower = path.Replace('\\', '/').ToLowerInvariant();
                foreach (var k in keywords)
                {
                    if (string.IsNullOrEmpty(k))
                        continue;
                    if (lower.Contains(k.ToLowerInvariant()))
                        return true;
                }
                return false;
            }
    
            private static string SanitizeName(string name)
            {
                foreach (var c in Path.GetInvalidFileNameChars())
                    name = name.Replace(c, '_');
                return name;
            }
    
            private static string ToLogicalPath(string absolutePath)
            {
                if (string.IsNullOrEmpty(absolutePath))
                    return absolutePath;
    
                string p = absolutePath.Replace('\\', '/');
                string lower = p.ToLowerInvariant();
                const string resourceTag = "/resource/";
                const string assetsTag = "/assets/";
    
                int idx = lower.IndexOf(resourceTag, StringComparison.Ordinal);
                if (idx >= 0)
                    return "Resource/" + p.Substring(idx + resourceTag.Length);
    
                idx = lower.IndexOf(assetsTag, StringComparison.Ordinal);
                if (idx >= 0)
                    return "Assets/" + p.Substring(idx + assetsTag.Length);
    
                return p;
            }
        }
    }
    ```

1. 엔진으로 가져오는 자동화 윈도우 스크립트를 짠다.
- 스크립트
    
    RelocateUnityExportResources.ps1 라는 이름으로 아래의 내용을 저장하세요
    
    ```csharp
    param(
        [string]$SourceRoot = "D:\Github\AliceRenderer\Assets\VFX\UnityExport",
        [string]$DestRoot = "D:\Github\AliceRenderer\Resource\VFX\UnityExport",
        [switch]$UpdateAssetJson = $true,
        [switch]$UpdateMaterialJson = $true
    )
    
    Set-StrictMode -Version Latest
    $ErrorActionPreference = "Stop"
    
    if (!(Test-Path -Path $SourceRoot)) {
        throw "SourceRoot not found: $SourceRoot"
    }
    
    if (!(Test-Path -Path $DestRoot)) {
        New-Item -ItemType Directory -Path $DestRoot | Out-Null
    }
    
    function Replace-ResourcesPath([string]$text, [string]$replacementPrefix) {
        $pattern = [regex]::Escape('"resources/')
        $repl = '"' + $replacementPrefix
        return [regex]::Replace($text, $pattern, $repl)
    }
    
    function Update-JsonFile([string]$filePath, [string]$replacementPrefix) {
        if (!(Test-Path -Path $filePath)) { return }
        $content = Get-Content -Raw -Path $filePath
        $updated = Replace-ResourcesPath -text $content -replacementPrefix $replacementPrefix
        if ($updated -ne $content) {
            Set-Content -Path $filePath -Value $updated -Encoding UTF8
        }
    }
    
    $effectDirs = Get-ChildItem -Path $SourceRoot -Directory
    
    foreach ($dir in $effectDirs) {
        $effectName = $dir.Name
        $srcEffect = $dir.FullName
        $srcResources = Join-Path $srcEffect "resources"
    
        $destEffect = Join-Path $DestRoot $effectName
        $destResources = Join-Path $destEffect "resources"
    
        if (Test-Path -Path $srcResources) {
            if (!(Test-Path -Path $destResources)) {
                New-Item -ItemType Directory -Path $destResources | Out-Null
            }
            Copy-Item -Path $srcResources\* -Destination $destResources -Recurse -Force
        }
    
        $replacementPrefix = "Resource/VFX/UnityExport/$effectName/resources/"
    
        if ($UpdateAssetJson) {
            $effectJson = Join-Path $srcEffect "effect.json"
            Update-JsonFile -filePath $effectJson -replacementPrefix $replacementPrefix
        }
    
        if ($UpdateMaterialJson -and (Test-Path -Path $destResources)) {
            $matDir = Join-Path $destResources "materials"
            if (Test-Path -Path $matDir) {
                Get-ChildItem -Path $matDir -Filter "*.mat.json" -File | ForEach-Object {
                    Update-JsonFile -filePath $_.FullName -replacementPrefix $replacementPrefix
                }
            }
        }
    }
    
    Write-Host "UnityExport resources relocated."
    ```

        # Bloom Post-Processing 시스템 문서
        
        ## 개요
        
        이 문서는 Alice Renderer에 구현된 Bloom post-processing 시스템과 DirectX11 post-processing pipeline 리팩터링에 대한 기술 문서입니다.
        
        ## 1. Bloom Post-Processing 시스템 구현
        
        ### 1.1 데이터 구조
        
        #### BloomSettings 구조체 (`src/Rendering/RenderTypes.h`)
        
        ```cpp
        struct BloomSettings
        {
            bool enabled = true;          // Bloom 활성화
            float intensity = 1.2f;      // 합성 강도
            float threshold = 1.0f;      // 밝기 추출 기준
            float knee = 0.5f;            // Soft threshold (0~1)
            float radius = 6.0f;          // Blur 크기 (sigma)
            int downsample = 2;           // 다운샘플링 (1=원본, 2=1/2, 4=1/4, 8=1/8, 16=1/16, 32=1/32, 64=1/64)
            float clamp = 10.0f;          // Bloom 값 상한 (옵션)
            int blurTaps = 9;              // Blur 탭 수 (5/7/9 등, 옵션)
        };
        ```
        
        #### BloomCB 구조체 (`src/Rendering/RenderTypes.h`)
        
        ```cpp
        struct BloomCB
        {
            float threshold;
            float knee;
            float intensity;
            float radius;
            DirectX::XMFLOAT2 texelSize;
            int downsample;
            float padding;
        };
        ```
        
        ### 1.2 리소스 구조
        
        Bloom 시스템은 다음 GPU 리소스를 사용합니다:
        
        - **Bright Pass 텍스처**: 밝은 영역 추출용 (다운샘플링 적용)
        - **Blur Ping-Pong 버퍼**: 가우시안 블러용 (2개의 텍스처로 ping-pong)
        - **HDR 합성 결과 텍스처**: Scene + Bloom 합성 결과 저장용 (`m_hdrAfterBloomRTV/SRV`)
        
        ### 1.3 렌더링 파이프라인
        
        Bloom 렌더링은 다음 단계로 구성됩니다:
        
        1. **Bright Pass**: HDR 씬에서 밝은 영역만 추출
        2. **Blur Pass (Horizontal)**: 수평 방향 가우시안 블러
        3. **Blur Pass (Vertical)**: 수직 방향 가우시안 블러
        4. **Composite Pass**: Scene + Bloom 합성 (HDR 결과, ToneMapping 없음)
        
        ## 2. Inspector UI 통합
        
        ### 2.1 UI 컨트롤
        
        `src/Editor/EditorCore.cpp`의 Inspector 창에 다음 Bloom 파라미터 컨트롤이 추가되었습니다:
        
        - **Enable Bloom**: Bloom 효과 활성화/비활성화 체크박스
        - **Intensity**: 합성 강도 슬라이더 (0.0 ~ 5.0)
        - **Threshold**: 밝기 추출 기준 슬라이더 (0.0 ~ 5.0)
        - **Knee**: Soft threshold 슬라이더 (0.0 ~ 1.0)
        - **Radius**: Blur 크기 슬라이더 (0.0 ~ 20.0)
        - **Downsample**: 다운샘플링 비율 드롭다운 (1x, 2x, 4x, 8x, 16x, 32x, 64x)
        - **Clamp**: Bloom 값 상한 슬라이더 (1.0 ~ 20.0)

        ## 3. 다운샘플링 개선
        
        ### 3.1 문제점 해결
        
        - **1/4 다운샘플링 문제**: 다운샘플링 크기 계산 시 최소 크기 보장 및 0으로 나누기 방지
        - **리소스 재생성**: 다운샘플링 값 변경 시 자동으로 Bloom 리소스 재생성
        
        ### 3.2 다운샘플링 확장
        
        다운샘플링 옵션을 1/64까지 확장:
        - 1x (원본)
        - 2x (1/2)
        - 4x (1/4)
        - 8x (1/8)
        - 16x (1/16)
        - 32x (1/32)
        - 64x (1/64)
        
        ### 3.3 안전성 개선
        
        - 다운샘플링 크기가 너무 작을 때 경고 메시지 출력
        - 텍셀 크기 계산 시 0으로 나누기 방지
        - 최소 크기 보장 (최소 1x1 텍스처 유지)
        
        ## 4. Post-Processing Pipeline 리팩터링
        
        ### 4.1 목표
        
        Bloom ON/OFF에 따라 OMSetRenderTargets 흐름을 명확히 관리하면서, ToneMapping은 항상 동일한 방식으로 호출되도록 구조화.
        
        ### 4.2 핵심 변경사항
        
        #### RenderToneMapping 시그니처 변경
        
        **변경 전:**
        ```cpp
        void RenderToneMapping(ID3D11RenderTargetView* targetRTV, const D3D11_VIEWPORT& viewport);
        ```
        
        **변경 후:**
        ```cpp
        void RenderToneMapping(ID3D11ShaderResourceView* inputSRV, ID3D11RenderTargetView* targetRTV, const D3D11_VIEWPORT& viewport);
        ```
        
        #### RenderPostProcess 함수 추가
        
        ```cpp
        void RenderPostProcess(const D3D11_VIEWPORT& viewport)
        {
            ID3D11ShaderResourceView* toneMapInputSRV = m_sceneColorSRV.Get(); // 기본값: SceneColor
        
            // Bloom ON: Bloom 패스 실행 → HDR 합성 결과를 toneMapInputSRV로 설정
            if (m_bloomSettings.enabled)
            {
                if (m_hdrAfterBloomRTV && m_bloomBrightPassPS)
                {
                    // Bloom 패스: SceneColor → Bloom 처리 → HDR 합성 (m_hdrAfterBloomRTV에 출력)
                    RenderBloomPass(m_sceneColorSRV.Get(), m_hdrAfterBloomRTV.Get(), viewport);
                    
                    // ToneMapping 입력을 HDR 합성 결과로 변경
                    toneMapInputSRV = m_hdrAfterBloomSRV.Get();
                }
            }
            // Bloom OFF: toneMapInputSRV는 그대로 m_sceneColorSRV 사용
        
            // ToneMapping 패스: 항상 실행 (입력만 Bloom ON/OFF에 따라 다름)
            RenderToneMapping(toneMapInputSRV, m_viewportRTV.Get(), viewport);
        }
        ```
        
        ### 4.3 렌더링 흐름
        
        #### Bloom OFF 시나리오
        
        1. `RenderPostProcess()` 호출
        2. `toneMapInputSRV = m_sceneColorSRV` (기본값)
        3. `RenderToneMapping(m_sceneColorSRV, viewportRTV, viewport)` 호출
        4. `OMSetRenderTargets(viewportRTV)` → ToneMapping 적용
        
        #### Bloom ON 시나리오
        
        1. `RenderPostProcess()` 호출
        2. `RenderBloomPass(m_sceneColorSRV, m_hdrAfterBloomRTV, viewport)` 호출
           - **Bright Pass**: `OMSetRenderTargets(m_bloomBrightRTV)`
           - **Blur H**: `OMSetRenderTargets(m_bloomBlurRTV[0])`
           - **Blur V**: `OMSetRenderTargets(m_bloomBlurRTV[1])`
           - **Composite**: `OMSetRenderTargets(m_hdrAfterBloomRTV)` → HDR 합성
        3. `toneMapInputSRV = m_hdrAfterBloomSRV` (HDR 합성 결과)
        4. `RenderToneMapping(m_hdrAfterBloomSRV, viewportRTV, viewport)` 호출
        5. `OMSetRenderTargets(viewportRTV)` → ToneMapping 적용
        
        ### 4.4 SRV/RTV 충돌 방지
        
        DirectX11에서는 동일 리소스를 RTV로 바인딩한 상태에서 SRV로 동시에 읽을 수 없습니다. 이를 방지하기 위해 각 패스 전환마다 SRV를 언바인드합니다:
        
        ```cpp
        // 패스 전환 시 SRV 언바인드
        ID3D11ShaderResourceView* nullSRVs[8] = { nullptr, nullptr, nullptr, nullptr, nullptr, nullptr, nullptr, nullptr };
        m_context->PSSetShaderResources(0, 8, nullSRVs);
        ```
        
        언바인드가 필요한 시점:
        - Bright Pass 완료 후
        - Horizontal Blur 완료 후
        - Vertical Blur 완료 후
        - Composite 완료 후
        - ToneMapping 완료 후
        
        ### 4.5 Bloom Composite 셰이더 수정
        
        **변경 전**: ToneMapping까지 포함
        ```hlsl
        float3 combined = sceneColor + bloomColor * g_Intensity;
        // Tone mapping
        float exposureFactor = pow(2.0f, g_Exposure);
        combined *= exposureFactor;
        float3 tonemapped = ACESFilm(combined);
        float3 final = LinearToSRGB(tonemapped);
        return float4(final, 1.0f);
        ```
        
        **변경 후**: HDR 합성만 수행
        ```hlsl
        float3 sceneColor = g_SceneHDR.Sample(g_SamplerLinear, input.uv).rgb;
        float3 bloomColor = g_Bloom.Sample(g_SamplerLinear, input.uv).rgb;
        
        // HDR 합성만 수행 (ToneMapping은 별도 패스에서 수행)
        float3 combined = sceneColor + bloomColor * g_Intensity;
        
        return float4(combined, 1.0f);
        ```

        ## 7. 성능 고려사항
        
        ### 7.1 다운샘플링
        
        - 다운샘플링 비율이 높을수록 성능이 향상되지만 품질이 저하됨
        - 권장 설정: 2x 또는 4x (품질과 성능의 균형)
        
        ### 7.2 Blur 탭 수
        
        - 현재 구현: 9-tap 가우시안 블러
        - 더 많은 탭을 사용하면 더 부드러운 블러가 가능하지만 성능이 저하됨
        
        ### 7.3 리소스 재생성
        
        - 다운샘플링 값 변경 시 리소스가 자동으로 재생성됨
        - 빈번한 변경은 성능에 영향을 줄 수 있으므로 주의
        
        ## 8. 문제 해결
        
        ### 8.1 Bloom이 보이지 않는 경우
        
        1. `enabled`가 `true`인지 확인
        2. `intensity`가 0보다 큰지 확인
        3. `threshold`가 너무 높지 않은지 확인 (씬이 충분히 밝은지 확인)
        4. 다운샘플링 값이 너무 크지 않은지 확인
        
        ### 8.2 성능 문제
        
        1. 다운샘플링 비율 증가 (2x → 4x → 8x)
        2. `radius` 값 감소
        3. Bloom 비활성화 (`enabled = false`)
        
        ### 8.3 컴파일 오류
        
        - `RenderToneMapping` 시그니처 변경으로 인한 호출부 수정 필요
        - `Engine.cpp`에서 `DeferredRenderSystem::RenderToneMapping` 호출 시 `inputSRV` 파라미터 추가 필요
