- 루트 모션 잠금 애니메이션으로 변환하기
    
    무기 애니메이션의 루트 모션이 캐릭터 애니와 중복 적용되어 싱크가 어긋나는 문제를 해결했습니다
    
    목표는 무기 루트의 이동을 제거하되 자식 본의 상대 좌표와 회전을 보존하는 것이었습니다.
    
    파이썬 코드로 본들의 좌표와 회전 변경을 시도하면서 코드를 바꿔가면서 했습니다.
    
    루트본을 0,0,0으로 옮기고 상대 위치로 무기본을 움직이는 걸로 키프레임을 옮겼습니다
    
    무기의 메시와 본은 x축으로 1.46579m 만큼 이동해 있는걸 확인했습니다.

    > **문제**
    > 
    - 무기 FBX는 루트 이동이 포함되어 있어 캐릭터 루트와 합치면 이중 이동이 발생했습니다.
    - 손 소켓 부착으로는 회전 정밀도가 맞지 않아 별도의 무기 애니를 유지해야 합니다.
    - 루트본이 실제로 키를 갖지 않고, 상위 체인/컨스트레인트 결과로 이동되는 구조였습니다.
    
    > 확인한 내용
    > 
    - Action Editor와 F‑Curve를 스캔하는 스크립트를 작성해 키 존재 여부를 확인했습니다
    - 루트본에 location 키가 없고, 이동은 부모 컨트롤 체인에서 발생함을 로그로 확인했습니다.
    - 월드 범위 로그로 실제 이동 축(Y)을 확정했습니다.
    
    > **시도했던 것**
    > 
    - 루트본 위치 키를 제거했으나 변화 없었습니다
    - 오브젝트 이동 보정 방식도 루트 이동 제거가 되지 않았습니다.
    - 모든 본을 베이크하여 역이동을 적용하자 회전이 꼬이고 상대 좌표가 깨졌습니다.

    - **어떤 액션에 어떤 F‑Curve가 있는지**를 **Text Editor에 로그로 출력해보자**

            ```csharp
            import bpy
            from collections import defaultdict
            
            ROOT_BONE = "root_weapon"
            LOG_TEXT = "root_lock_debug"
            
            def log(line):
                txt = bpy.data.texts.get(LOG_TEXT)
                if not txt:
                    txt = bpy.data.texts.new(LOG_TEXT)
                txt.write(line + "\n")
                print(line)
            
            def get_active_armature():
                if bpy.context.active_object and bpy.context.active_object.type == "ARMATURE":
                    return bpy.context.active_object
                for obj in bpy.context.selected_objects:
                    if obj.type == "ARMATURE":
                        return obj
                return None
            
            def action_summary(action):
                root_prefix = f'pose.bones["{ROOT_BONE}"]'
                stats = {
                    "root_loc": [],
                    "root_rot": [],
                    "obj_loc": [],
                    "any_loc_bones": defaultdict(int),
                }
            
                for fc in action.fcurves:
                    dp = fc.data_path
            
                    if dp == "location":
                        stats["obj_loc"].append(fc)
                        continue
            
                    if dp.startswith('pose.bones["'):
                        # bone name extract
                        bone_name = dp.split('"')[1]
                        if dp.endswith('"].location'):
                            stats["any_loc_bones"][bone_name] += 1
            
                        if dp.startswith(root_prefix):
                            if dp.endswith('"].location'):
                                stats["root_loc"].append(fc)
                            elif dp.endswith('"].rotation_euler') or dp.endswith('"].rotation_quaternion'):
                                stats["root_rot"].append(fc)
            
                return stats
            
            def fc_info(fc):
                keys = fc.keyframe_points
                if not keys:
                    return "keys=0"
                vals = [k.co.y for k in keys]
                return f"keys={len(keys)}, min={min(vals):.6f}, max={max(vals):.6f}"
            
            def run():
                txt = bpy.data.texts.get(LOG_TEXT)
                if not txt:
                    txt = bpy.data.texts.new(LOG_TEXT)
                txt.clear()
            
                arm = get_active_armature()
                if not arm:
                    log("No active/selected Armature.")
                    return
            
                log(f"Active Armature: {arm.name}")
                log(f"Has root bone '{ROOT_BONE}': {ROOT_BONE in arm.pose.bones}")
            
                ad = arm.animation_data
                if not ad:
                    log("No animation_data on armature.")
                    return
            
                if ad.action:
                    log(f"Active Action: {ad.action.name}")
                else:
                    log("Active Action: None")
            
                # Actions connected to this armature (Active + NLA)
                actions = []
                if ad.action:
                    actions.append(ad.action)
                for track in ad.nla_tracks:
                    for strip in track.strips:
                        if strip.action and strip.action not in actions:
                            actions.append(strip.action)
            
                if not actions:
                    log("No actions found in Active/NLA.")
                    return
            
                log(f"Actions to inspect: {len(actions)}")
                for act in actions:
                    log(f"--- Action: {act.name} ---")
                    stats = action_summary(act)
            
                    if stats["root_loc"]:
                        for fc in stats["root_loc"]:
                            log(f"root loc axis {fc.array_index}: {fc_info(fc)}")
                    else:
                        log("root loc curves: NONE")
            
                    if stats["root_rot"]:
                        for fc in stats["root_rot"]:
                            log(f"root rot axis {fc.array_index}: {fc_info(fc)}")
                    else:
                        log("root rot curves: NONE")
            
                    if stats["obj_loc"]:
                        for fc in stats["obj_loc"]:
                            log(f"object loc axis {fc.array_index}: {fc_info(fc)}")
                    else:
                        log("object loc curves: NONE")
            
                    if stats["any_loc_bones"]:
                        top = sorted(stats["any_loc_bones"].items(), key=lambda x: -x[1])
                        log("bones with location curves:")
                        for name, count in top[:15]:
                            log(f"  {name}: {count}")
                    else:
                        log("no bone location curves found at all.")
            
                log("Done. See Text datablock: root_lock_debug")
            
            run()
            
            ```

    - **root_weapon 월드 이동을 역이동으로 상쇄해보자**
        - 코드
            
            ```csharp
            import bpy
            from mathutils import Matrix, Vector
            
            ROOT_BONE = "root_weapon"
            AXES = (False, False, True)  # Z축만 제거. (X,Y,Z) 전부 제거하려면 (True, True, True)
            LOG_TEXT = "root_lock_apply_log"
            
            def log(line):
                txt = bpy.data.texts.get(LOG_TEXT) or bpy.data.texts.new(LOG_TEXT)
                txt.write(line + "\n")
                print(line)
            
            def popup(msg):
                def draw(self, context):
                    self.layout.label(text=msg)
                bpy.context.window_manager.popup_menu(draw, title="Root Lock", icon='INFO')
            
            def get_armature():
                if bpy.context.active_object and bpy.context.active_object.type == "ARMATURE":
                    return bpy.context.active_object
                for obj in bpy.context.selected_objects:
                    if obj.type == "ARMATURE":
                        return obj
                return None
            
            def run():
                txt = bpy.data.texts.get(LOG_TEXT) or bpy.data.texts.new(LOG_TEXT)
                txt.clear()
            
                arm = get_armature()
                if not arm:
                    log("No active/selected Armature.")
                    popup("No Armature selected")
                    return
            
                if ROOT_BONE not in arm.pose.bones:
                    log(f"Missing bone: {ROOT_BONE}")
                    popup("Missing root bone")
                    return
            
                ad = arm.animation_data
                if not ad or not ad.action:
                    log("No active action.")
                    popup("No active action")
                    return
            
                action = ad.action
                scene = bpy.context.scene
                start, end = map(int, action.frame_range)
                frames = range(start, end + 1)
            
                log(f"Active Armature: {arm.name}")
                log(f"Active Action: {action.name}")
                log(f"Frames: {start}-{end}")
            
                # 1) 샘플링
                samples = {}
                minv = Vector((1e9, 1e9, 1e9))
                maxv = Vector((-1e9, -1e9, -1e9))
            
                cur = scene.frame_current
                for f in frames:
                    scene.frame_set(f)
                    bpy.context.view_layer.update()
            
                    root_world = arm.matrix_world @ arm.pose.bones[ROOT_BONE].matrix
                    t = root_world.to_translation()
            
                    minv.x = min(minv.x, t.x); minv.y = min(minv.y, t.y); minv.z = min(minv.z, t.z)
                    maxv.x = max(maxv.x, t.x); maxv.y = max(maxv.y, t.y); maxv.z = max(maxv.z, t.z)
            
                    t_mask = Vector((
                        t.x if AXES[0] else 0.0,
                        t.y if AXES[1] else 0.0,
                        t.z if AXES[2] else 0.0
                    ))
            
                    orig_world = arm.matrix_world.copy()
                    new_world = Matrix.Translation(-t_mask) @ orig_world
            
                    # 로컬 위치를 얻기 위해 잠깐 적용
                    arm.matrix_world = new_world
                    new_loc = arm.location.copy()
                    arm.matrix_world = orig_world
            
                    samples[f] = new_loc
            
                log(f"root_weapon world range:")
                log(f"  X: {minv.x:.6f} .. {maxv.x:.6f}")
                log(f"  Y: {minv.y:.6f} .. {maxv.y:.6f}")
                log(f"  Z: {minv.z:.6f} .. {maxv.z:.6f}")
            
                # 2) 오브젝트 location F‑Curve에 키 생성
                for axis, use in enumerate(AXES):
                    if not use:
                        continue
                    fc = action.fcurves.find("location", index=axis)
                    if not fc:
                        fc = action.fcurves.new("location", index=axis, action_group="Object")
                    fc.keyframe_points.clear()
                    for f in frames:
                        v = samples[f][axis]
                        fc.keyframe_points.insert(f, v, options={'FAST'})
                    fc.update()
            
                scene.frame_set(cur)
                bpy.context.view_layer.update()
            
                log("Done. Check Text: root_lock_apply_log")
                popup("Done. Check root_lock_apply_log")
            
            run()
            
            ```

    - **디버그 스크립트 (액션/루트 이동 확인)를 띄워서 부모/컨스트레인트 체인에서 들어오는 본인지를 판단해보자**
        - 코드
            
            ```csharp
            import bpy
            from mathutils import Matrix, Vector
            
            ROOT_BONE = "root_weapon"
            LOCK_AXES = "AUTO"   # "AUTO" 또는 (True, True, True) 등
            LOCK_MODE = "ABSOLUTE"  # "ABSOLUTE"=원점 고정, "RELATIVE"=첫 프레임 기준
            DUPLICATE_ACTION = True
            ACTION_SUFFIX = "_rootLocked"
            EPS = 1e-6
            LOG_TEXT = "root_lock_apply_log"
            
            def log(line):
                txt = bpy.data.texts.get(LOG_TEXT) or bpy.data.texts.new(LOG_TEXT)
                txt.write(line + "\n")
                print(line)
            
            def popup(msg):
                def draw(self, context):
                    self.layout.label(text=msg)
                bpy.context.window_manager.popup_menu(draw, title="Root Lock", icon='INFO')
            
            def get_armature():
                if bpy.context.active_object and bpy.context.active_object.type == "ARMATURE":
                    return bpy.context.active_object
                for obj in bpy.context.selected_objects:
                    if obj.type == "ARMATURE":
                        return obj
                return None
            
            def get_action_from_editor():
                for area in bpy.context.window.screen.areas:
                    if area.type == 'DOPESHEET_EDITOR':
                        space = area.spaces.active
                        if space.mode == 'ACTION' and space.action:
                            return space.action, f"Action Editor (pinned={space.use_pin})"
                return None, ""
            
            def run():
                txt = bpy.data.texts.get(LOG_TEXT) or bpy.data.texts.new(LOG_TEXT)
                txt.clear()
            
                arm = get_armature()
                if not arm:
                    log("No active/selected Armature.")
                    popup("No Armature selected")
                    return
            
                act, src = get_action_from_editor()
                if not act:
                    if arm.animation_data and arm.animation_data.action:
                        act = arm.animation_data.action
                        src = "Armature Active Action"
                    else:
                        log("No action found.")
                        popup("No action found")
                        return
            
                log(f"Using Action: {act.name} (source: {src})")
            
                if ROOT_BONE not in arm.pose.bones:
                    log(f"Missing bone: {ROOT_BONE}")
                    popup("Missing root bone")
                    return
            
                if DUPLICATE_ACTION:
                    act = act.copy()
                    act.name = act.name + ACTION_SUFFIX
                    log(f"Duplicated action -> {act.name}")
            
                if not arm.animation_data:
                    arm.animation_data_create()
                arm.animation_data.action = act
            
                scene = bpy.context.scene
                start, end = map(int, act.frame_range)
                frames = range(start, end + 1)
            
                # 샘플링 (root head world)
                samples = {}
                minv = Vector((1e9,1e9,1e9))
                maxv = Vector((-1e9,-1e9,-1e9))
            
                cur = scene.frame_current
                for f in frames:
                    scene.frame_set(f)
                    bpy.context.view_layer.update()
            
                    pb = arm.pose.bones[ROOT_BONE]
                    head_w = arm.matrix_world @ pb.head
            
                    minv = Vector((min(minv.x, head_w.x), min(minv.y, head_w.y), min(minv.z, head_w.z)))
                    maxv = Vector((max(maxv.x, head_w.x), max(maxv.y, head_w.y), max(maxv.z, head_w.z)))
            
                    samples[f] = head_w.copy()
            
                rng = Vector((maxv.x - minv.x, maxv.y - minv.y, maxv.z - minv.z))
                log(f"root head world range: X={rng.x:.6f}, Y={rng.y:.6f}, Z={rng.z:.6f}")
            
                if LOCK_AXES == "AUTO":
                    axes = (rng.x > EPS, rng.y > EPS, rng.z > EPS)
                else:
                    axes = LOCK_AXES
            ```

우리 엔진의 트랜스폼  

> 위치
> 

왼손 좌표계, Y-Up, Z-Forward

> 회전
> 
- `v.x` = **Pitch** (X축 회전, 고개 끄덕임)
- `v.y` = **Yaw** (Y축 회전, 좌우 도리도리)
- `v.z` = **Roll** (Z축 회전, 갸우뚱)

> 애니메이션 시간 단위
> 
- 엔진 내부의 모든 시간은 **초(Seconds)** 단위입니다.
- Assimp에서 가져온 틱(Tick)을 초 단위로 변환해서 씁니다.
- 블렌더에서 30FPS로 만든 60프레임짜리 애니메이션이라면, 엔진에서는 **2.0초**(`0.0f ~ 2.0f`)로 인식됩니다
- 예를들어 **따라서:** `anim->base.timeA`에 들어가는 값은 2.5초 애니메이션이면 `2.5`가 끝입니다

> 본 개수
> 
- 80 ~ 200

> 버텍스 개수
> 
- 3만개정도

> 텍스쳐 사이즈
> 
- 캐릭터는 2K까지 가능. 배경은 512 ~ 1k

> 엔진에서의 FBX 다루기

- fbx 파일 사이즈 전부 바꾸는 스크립트
    
    ```cpp
    "C:\Program Files\Blender Foundation\Blender 4.5\blender.exe" -b -P batch_fbx_fix.py -- "D:\Github\AliceRenderer\Resource\fbx" --inplace
    ```
    
    ```python
    # batch_fbx_fix.py
    # Usage:
    #   blender -b -P batch_fbx_fix.py -- <input_root> [output_root] [--inplace] [--suffix=_fixed] [--apply_scale] [--apply_armature_scale]
    #
    # Example (safe, outputs to a new folder):
    #   blender -b -P batch_fbx_fix.py -- "D:\Github\AliceRenderer\Resource\fbx" "D:\Github\AliceRenderer\Resource\fbx_fixed"
    #
    # Example (overwrite in place - be careful):
    #   blender -b -P batch_fbx_fix.py -- "D:\Github\AliceRenderer\Resource\fbx" --inplace
    
    import bpy
    import os
    import sys
    import traceback
    
    def parse_args():
        argv = sys.argv
        argv = argv[argv.index("--") + 1:] if "--" in argv else []
    
        input_root = os.getcwd()
        output_root = None
        inplace = False
        suffix = "_fixed"
        apply_scale = True
        apply_armature_scale = False
    
        # positional
        pos = []
        i = 0
        while i < len(argv):
            a = argv[i]
            if a.startswith("--"):
                if a == "--inplace":
                    inplace = True
                elif a.startswith("--suffix="):
                    suffix = a.split("=", 1)[1]
                elif a == "--apply_scale":
                    apply_scale = True
                elif a == "--no_apply_scale":
                    apply_scale = False
                elif a == "--apply_armature_scale":
                    apply_armature_scale = True
                else:
                    print(f"[WARN] Unknown option: {a}")
            else:
                pos.append(a)
            i += 1
    
        if len(pos) >= 1:
            input_root = pos[0]
        if len(pos) >= 2:
            output_root = pos[1]
    
        input_root = os.path.abspath(input_root)
    
        if inplace:
            output_root = input_root
        else:
            if output_root is None:
                output_root = input_root.rstrip("\\/") + "_fixed"
            output_root = os.path.abspath(output_root)
    
        return input_root, output_root, inplace, suffix, apply_scale, apply_armature_scale
    
    def reset_scene():
        # 완전 초기화 (파일마다 누적 방지)
        bpy.ops.wm.read_factory_settings(use_empty=True)
        # 모드 보정
        if bpy.ops.object.mode_set.poll():
            try:
                bpy.ops.object.mode_set(mode='OBJECT')
            except:
                pass
    
    def import_fbx(path):
        # 필요하면 import 옵션 추가 가능
        bpy.ops.import_scene.fbx(
            filepath=path,
            use_image_search=True
        )
    
    def apply_scale_to_objects(apply_armature_scale=False):
        # Ctrl-A 중 "Scale만" 적용 (위치/회전은 건드리지 않음)
        if not bpy.ops.object.mode_set.poll():
            return
        bpy.ops.object.mode_set(mode='OBJECT')
    
        bpy.ops.object.select_all(action='DESELECT')
    
        selectable_types = {"MESH", "EMPTY", "CAMERA", "LIGHT"}
        if apply_armature_scale:
            selectable_types.add("ARMATURE")
    
        selected = []
        for obj in bpy.context.scene.objects:
            if obj.type in selectable_types:
                # Armature는 애니메이션/리깅 깨질 수 있어서 기본은 제외
                obj.select_set(True)
                selected.append(obj)
    
        if selected:
            bpy.context.view_layer.objects.active = selected[0]
            bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
    
    def export_fbx(path):
        os.makedirs(os.path.dirname(path), exist_ok=True)
    
        bpy.ops.export_scene.fbx(
            filepath=path,
            use_selection=False,
            use_active_collection=False,
    
            # 스케일 관련 (요청사항)
            global_scale=1.0,
            apply_unit_scale=True,
            apply_scale_options='FBX_SCALE_UNITS',  # Apply Scalings = FBX Units Scale
    
            # 축/트랜스폼 (스크린샷 기준)
            axis_forward='-Z',
            axis_up='Y',
            use_space_transform=True,
            bake_space_transform=True,  # Transform > Apply Transform
    
            # 오브젝트 타입
            object_types={'EMPTY', 'CAMERA', 'LIGHT', 'ARMATURE', 'MESH', 'OTHER'},
    
            # Armature Leaf Bones (요청사항)
            add_leaf_bones=False,
    
            # Bone axis (스크린샷 기본값)
            primary_bone_axis='Y',
            secondary_bone_axis='X',
    
            # 나머진 기본값에 가깝게
            use_mesh_modifiers=True,
            use_custom_props=False,
            path_mode='AUTO',
            embed_textures=False,
        )
    
    def iter_fbx_files(root):
        for dirpath, _, filenames in os.walk(root):
            for fn in filenames:
                if fn.lower().endswith(".fbx"):
                    yield os.path.join(dirpath, fn)
    
    def main():
        input_root, output_root, inplace, suffix, apply_scale, apply_armature_scale = parse_args()
        print("=== Batch FBX Fix ===")
        print("Input :", input_root)
        print("Output:", output_root)
        print("Inplace:", inplace)
        print("Suffix:", suffix)
        print("Apply Scale:", apply_scale, "(armature:", apply_armature_scale, ")")
        print("=====================")
    
        files = list(iter_fbx_files(input_root))
        print(f"Found {len(files)} FBX files.")
    
        errors = []
    
        for idx, in_path in enumerate(files, 1):
            try:
                rel = os.path.relpath(in_path, input_root)
                base, ext = os.path.splitext(rel)
    
                if inplace:
                    out_path = in_path  # 덮어쓰기 (주의)
                else:
                    out_rel = base + suffix + ext
                    out_path = os.path.join(output_root, out_rel)
    
                print(f"\n[{idx}/{len(files)}] {in_path}")
                print(f" -> {out_path}")
    
                reset_scene()
                import_fbx(in_path)
    
                if apply_scale:
                    apply_scale_to_objects(apply_armature_scale=apply_armature_scale)
    
                export_fbx(out_path)
                print("   OK")
    
            except Exception as e:
                print("   FAILED:", e)
                traceback.print_exc()
                errors.append((in_path, str(e)))
    
        print("\n=== Done ===")
        if errors:
            print(f"Errors: {len(errors)}")
            for p, msg in errors:
                print(" -", p, ":", msg)
        else:
            print("All files processed successfully.")
    
    if __name__ == "__main__":
        main()
    
    ```

- 텍스쳐 이미지로 노드 연결하기
    
    ![image.png](%EC%95%84%ED%8A%B8%20%ED%8A%9C%ED%86%A0%EB%A6%AC%EC%96%BC/image%2060.png)
    
    ```python
    import bpy
    import os
    
    def create_and_save_textures_to_disk():
        # 1. 현재 블렌더 파일 경로 확인 (저장 위치 기준)
        if not bpy.data.filepath:
            print("❌ 오류: 블렌더 파일을 먼저 저장해주세요! 파일 위치를 모르면 텍스처를 저장할 수 없습니다.")
            return
    
        # 2. 텍스처 저장할 폴더 만들기 (블렌더 파일 옆 'generated_textures' 폴더)
        base_path = os.path.dirname(bpy.data.filepath)
        save_dir = os.path.join(base_path, "generated_textures")
        
        if not os.path.exists(save_dir):
            os.makedirs(save_dir)
    
        # 설정
        IMG_WIDTH = 512
        IMG_HEIGHT = 512
        count = 0
    
        print(f"--- 텍스처 파일 저장 시작: {save_dir} ---")
    
        for mat in bpy.data.materials:
            if not mat.use_nodes:
                continue
    
            # Principled BSDF 찾기
            bsdf = None
            for node in mat.node_tree.nodes:
                if node.type == 'BSDF_PRINCIPLED':
                    bsdf = node
                    break
            
            if not bsdf:
                continue
    
            base_color_socket = bsdf.inputs['Base Color']
    
            # 이미 연결된 게 있으면 건너뛰기
            if base_color_socket.is_linked:
                continue
    
            # 색상 값 읽기
            color = base_color_socket.default_value
            r, g, b, a = color[0], color[1], color[2], 1.0
    
            # 3. 픽셀 데이터 생성
            pixels = [r, g, b, a] * (IMG_WIDTH * IMG_HEIGHT)
    
            # 4. 이름 설정 (아까 보신 그 이름 형식 유지)
            img_name = f"{mat.name}_BaseColor_Tex"
            file_name = f"{img_name}.png"
            full_filepath = os.path.join(save_dir, file_name)
    
            # 기존에 같은 이름의 이미지가 블렌더 내부에 있으면 삭제 (충돌 방지)
            if img_name in bpy.data.images:
                bpy.data.images.remove(bpy.data.images[img_name])
    
            # 5. 블렌더 이미지 데이터 생성
            image = bpy.data.images.new(name=img_name, width=IMG_WIDTH, height=IMG_HEIGHT, alpha=True)
            image.pixels = pixels
    
            # 6. 중요: 실제 하드디스크에 저장 (Save)
            image.filepath_raw = full_filepath
            image.file_format = 'PNG'
            try:
                image.save() # 여기서 실제 파일이 생깁니다!
            except:
                print(f"저장 실패: {mat.name}")
                continue
    
            # 7. 노드 연결
            nodes = mat.node_tree.nodes
            links = mat.node_tree.links
    
            tex_node = nodes.new('ShaderNodeTexImage')
            tex_node.image = image
            tex_node.location = (bsdf.location.x - 300, bsdf.location.y)
            
            links.new(tex_node.outputs['Color'], base_color_socket)
            
            count += 1
            print(f"저장됨: {file_name}")
    
        print(f"--- 총 {count}개 저장 및 연결 완료 ---")
    
    create_and_save_textures_to_disk()
    ```
